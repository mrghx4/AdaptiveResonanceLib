function(add_artlib_cpp_test target source)
    add_executable(${target} ${source})
    target_link_libraries(${target} PRIVATE artlib_cpp_core)
    artlib_cpp_apply_hardening(${target})
    add_test(NAME ${target} COMMAND ${target})
endfunction()
